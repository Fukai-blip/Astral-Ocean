      function deterministicFallbackHashHex(input) {

      function loadStateFromLocalStorage() {
        const fallback = createInitialState();
        try {
          const raw = localStorage.getItem(STORAGE_KEYS.STATE);
          if (!raw) return fallback;
          const parsed = JSON.parse(raw);
          if (!parsed || !Array.isArray(parsed.stars) || parsed.stars.length === 0 || !Array.isArray(parsed.planets)) {
            return fallback;
          }
          const validFilters = new Set(['all', 'document', 'image']);
          return {
            ...fallback,
            ...parsed,
            typeFilter: validFilters.has(parsed.typeFilter) ? parsed.typeFilter : 'all',
            collapsedStarIds: Array.isArray(parsed.collapsedStarIds) ? parsed.collapsedStarIds : [],
            expandedPlanetIds: Array.isArray(parsed.expandedPlanetIds) ? parsed.expandedPlanetIds : [],
            planets: normalizePlanetsList(parsed.planets)
          };
        } catch (err) {
          return fallback;
        }
      }

      function loadSyncConfigFromLocalStorage() {
        try {
          const raw = localStorage.getItem(STORAGE_KEYS.SYNC_CONFIG);
          if (!raw) return { ...DEFAULT_SUPABASE_CONFIG };
          const parsed = JSON.parse(raw);
          const url = normalizeSupabaseUrl(parsed && parsed.url ? parsed.url : DEFAULT_SUPABASE_CONFIG.url);
          const anonKey = (parsed && parsed.anonKey) ? String(parsed.anonKey).trim() : DEFAULT_SUPABASE_CONFIG.anonKey;
          return {
            enabled: parsed && typeof parsed.enabled === 'boolean' ? parsed.enabled : Boolean(url && anonKey),
            url: url || DEFAULT_SUPABASE_CONFIG.url,
            anonKey: anonKey || DEFAULT_SUPABASE_CONFIG.anonKey,
            table: (parsed && parsed.table) ? String(parsed.table).trim() : DEFAULT_SUPABASE_CONFIG.table,
            roomId: (parsed && parsed.roomId) ? String(parsed.roomId).trim() : DEFAULT_SUPABASE_CONFIG.roomId
          };
        } catch (err) {
          return { ...DEFAULT_SUPABASE_CONFIG };
        }
      }

      /**
       * 招待コードの検証と認証状態の更新（大文字/小文字・前後空白を正規化）
       * @param {string} rawCode
       * @param {boolean} showFeedback
       * @returns {Promise<boolean>}
       */
      async function verifyAndAuthenticateInviteCode(rawCode, showFeedback = true) {
        const normalizedCode = String(rawCode || '').trim().toUpperCase();
        if (!normalizedCode) {
          if (showFeedback) showToast('招待コードを入力してください', 'ph-warning');
          return false;
        }
        await expectedHashReadyPromise;
        const expected = EXPECTED_INVITE_HASH_HEX || deterministicFallbackHashHex(INVITE_CODE_PLAIN.trim().toUpperCase());
        const actual = await computeInviteHashHex(normalizedCode);
        if (actual && actual === expected) {
          appState.inviteAuth = {
            authenticated: true,
            codeHint: normalizedCode,
            verifiedAt: Date.now()
          };
          saveInviteAuthToLocalStorage();
          updateInviteAuthUI();
          if (appState.syncConfig && appState.syncConfig.enabled) {
            startSyncLoop();
            await pullStateFromSupabase(false);
          }
          if (showFeedback) {
            showToast('招待コード認証に成功しました。共有ルーム同期を有効化しました', 'ph-key');
          }
          return true;
        }
        if (showFeedback) {
          showToast(`招待コードが一致しません（${INVITE_CODE_PLAIN} を確認してください）`, 'ph-warning');
        }
        return false;
      }

      function buildSupabaseHeaders(apiKey, extra = {}) {
        const key = String(apiKey || '').trim();
        const headers = {
          'apikey': key,
          ...extra
        };
        if (/^eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$/.test(key)) {
          headers['Authorization'] = `Bearer ${key}`;
        }
        return headers;
      }

      async function pushStateToSupabase(notify = false) {
        if (!isInviteAuthenticated()) {
          stopSyncLoop();
          appState.isSyncing = false;
          updateInviteAuthUI();
          if (notify) showToast('招待コード未認証のためSupabase送信をブロックしました', 'ph-lock');
          return;
        }
        const cfg = appState.syncConfig;
        if (!cfg || !cfg.enabled || !cfg.url || !cfg.anonKey) return;
        if (appState.isSyncing) return;

        appState.isSyncing = true;
        updateSyncStatusUI('送信中...', 'bg-amber-400 animate-pulse');
        try {
          const baseUrl = normalizeSupabaseUrl(cfg.url);
          const table = encodeURIComponent((cfg.table || 'astral_ocean_rooms').trim());
          const roomId = (cfg.roomId || 'astral-main-room').trim();
          const ts = Number(appState.data.updatedAt || Date.now());
          appState.data.updatedAt = ts;

          const endpoint = `${baseUrl}/rest/v1/${table}?on_conflict=room_id`;
          const res = await fetch(endpoint, {
            method: 'POST',
            headers: buildSupabaseHeaders(cfg.anonKey, {
              'Content-Type': 'application/json',
              'Prefer': 'resolution=merge-duplicates,return=minimal'
            }),
            body: JSON.stringify([
              {
                room_id: roomId,
                payload: appState.data,
                updated_at: new Date(ts).toISOString()
              }
            ])
          });
          if (!res.ok) {
            const errText = await res.text().catch(() => '');
            throw new Error(`HTTP ${res.status}${errText ? `: ${errText.slice(0, 80)}` : ''}`);
          }
          appState.lastSyncedRemoteTs = ts;
          updateSyncStatusUI('Supabase同期済み', 'bg-emerald-400 shadow-[0_0_8px_#34d399]');
          if (notify) showToast('Supabaseへ星系データを保存しました', 'ph-cloud-arrow-up');
        } catch (err) {
          updateSyncStatusUI('同期エラー（テーブル未作成の可能性）', 'bg-rose-500');
          if (notify) showToast(`Supabase送信失敗: ${err.message}`, 'ph-warning');
        } finally {
          appState.isSyncing = false;
        }
      }

      async function pullStateFromSupabase(notify = false) {
        if (!isInviteAuthenticated()) {
          stopSyncLoop();
          appState.isSyncing = false;
          updateInviteAuthUI();
          if (notify) showToast('招待コード未認証のためSupabase同期をブロックしました', 'ph-lock');
          return;
        }
        const cfg = appState.syncConfig;
        if (!cfg || !cfg.enabled || !cfg.url || !cfg.anonKey) return;
        if (appState.isSyncing) return;

        appState.isSyncing = true;
        let shouldPushAfterRelease = false;
        try {
          const baseUrl = normalizeSupabaseUrl(cfg.url);
          const table = encodeURIComponent((cfg.table || 'astral_ocean_rooms').trim());
          const roomId = encodeURIComponent((cfg.roomId || 'astral-main-room').trim());
          const endpoint = `${baseUrl}/rest/v1/${table}?room_id=eq.${roomId}&select=payload,updated_at`;
          const res = await fetch(endpoint, {
            method: 'GET',
            headers: buildSupabaseHeaders(cfg.anonKey)
          });
          if (!res.ok) {
            const errText = await res.text().catch(() => '');
            throw new Error(`HTTP ${res.status}${errText ? `: ${errText.slice(0, 80)}` : ''}`);
          }
          const rows = await res.json();
          if (Array.isArray(rows) && rows.length > 0 && rows[0].payload) {
            const remote = rows[0].payload;
            const rowUpdatedMs = rows[0].updated_at ? new Date(rows[0].updated_at).getTime() : 0;
            const remoteTs = Math.max(Number(remote.updatedAt || 0), Number.isFinite(rowUpdatedMs) ? rowUpdatedMs : 0);
            const localTs = Number(appState.data.updatedAt || 0);

            if (remoteTs > localTs && Array.isArray(remote.stars) && remote.stars.length > 0 && Array.isArray(remote.planets)) {
              const validFilters = new Set(['all', 'document', 'image']);
              appState.data = {
                ...createInitialState(),
                ...remote,
                typeFilter: validFilters.has(remote.typeFilter) ? remote.typeFilter : 'all',
                collapsedStarIds: Array.isArray(remote.collapsedStarIds) ? remote.collapsedStarIds : [],
                expandedPlanetIds: Array.isArray(remote.expandedPlanetIds) ? remote.expandedPlanetIds : [],
                planets: normalizePlanetsList(remote.planets),
                updatedAt: remoteTs
              };
              appState.lastSyncedRemoteTs = remoteTs;
              saveStateToLocalStorage(false);
              reconcileSelection();
              rebuild3DScene();
              renderAllUI();
              updateSyncStatusUI('Supabase同期済み', 'bg-emerald-400 shadow-[0_0_8px_#34d399]');
              if (notify) showToast('Supabaseから最新の星系データを取得しました', 'ph-cloud-arrow-down');
            } else {
              updateSyncStatusUI('Supabase同期済み', 'bg-emerald-400 shadow-[0_0_8px_#34d399]');
              if (notify) {
                shouldPushAfterRelease = true;
              }
            }
          } else if (notify) {
            shouldPushAfterRelease = true;
          }
        } catch (err) {
          updateSyncStatusUI('同期エラー', 'bg-rose-500');
          if (notify) showToast(`Supabase同期確認: ${err.message}`, 'ph-warning');
        } finally {
          appState.isSyncing = false;
        }

        if (shouldPushAfterRelease) {
          await pushStateToSupabase(true);
        }
      }

      function stopSyncLoop() {
        if (appState.syncTimerId) {
          clearInterval(appState.syncTimerId);
          appState.syncTimerId = null;
        }
      }

      function startSyncLoop() {
        stopSyncLoop();
        if (!isInviteAuthenticated()) {
          updateInviteAuthUI();
          return;
        }
        if (!appState.syncConfig || !appState.syncConfig.enabled) return;
        pullStateFromSupabase(false);
        appState.syncTimerId = setInterval(() => {
          if (!isInviteAuthenticated()) {
            stopSyncLoop();
            updateInviteAuthUI();
            return;
          }
          pullStateFromSupabase(false);
        }, 12000);
      }

      function getStarById(starId) {
        return appState.data.stars.find((s) => s.id === starId) || null;
      }

      function getPlanetById(planetId) {
        return appState.data.planets.find((p) => p.id === planetId) || null;
      }

      function getPlanetsForStar(starId) {
        return appState.data.planets.filter((p) => p.starId === starId);
      }

      function getPlanetWorldPosition(planet, elapsedSeconds = 0) {
        const star = getStarById(planet.starId) || appState.data.stars[0];
        const basePos = star && star.position ? star.position : { x: 0, y: 0, z: 0 };
        const angle =
          Number(planet.orbitAngle || 0) +
          elapsedSeconds * Number(planet.orbitSpeed || 0.25) * Number(appState.data.orbitSpeed || 0.5);
        const r = Number(planet.orbitRadius || 40);
        const inc = Number(planet.inclination || 0);
        const x = (basePos.x || 0) + Math.cos(angle) * r;
        const z = (basePos.z || 0) + Math.sin(angle) * r;
        const y = (basePos.y || 0) + Math.sin(angle * 1.5) * Math.sin(inc) * r * 0.45;
        return { x, y, z, angle };
      }

      function reconcileSelection() {
        if (appState.data.selectedKind === 'star') {
          const found = getStarById(appState.data.selectedId);
          if (!found && appState.data.stars.length > 0) {
            appState.data.selectedId = appState.data.stars[0].id;
          }
        } else if (appState.data.selectedKind === 'planet') {
          const found = getPlanetById(appState.data.selectedId);
          if (!found) {
            appState.data.selectedKind = 'star';
            appState.data.selectedId = appState.data.stars[0] ? appState.data.stars[0].id : null;
          }
        }
        if (appState.data.lockedTarget) {
          if (appState.data.lockedTarget.kind === 'star' && !getStarById(appState.data.lockedTarget.id)) {
            appState.data.lockedTarget = null;
          } else if (appState.data.lockedTarget.kind === 'planet' && !getPlanetById(appState.data.lockedTarget.id)) {
            appState.data.lockedTarget = null;
          }
        }
      }

      function ensureThreeSetup() {
        if (!isThreeAvailable()) return false;
        if (viewport.renderer && viewport.scene && viewport.camera) return true;
        try {
          const width = Math.max(320, (els.viewportContainer && els.viewportContainer.clientWidth) || window.innerWidth || 1024);
          const height = Math.max(240, (els.viewportContainer && els.viewportContainer.clientHeight) || window.innerHeight || 768);

          viewport.raycaster = new THREE.Raycaster();
          viewport.pointerNDC = new THREE.Vector2();
          viewport.scene = new THREE.Scene();
          viewport.scene.background = new THREE.Color(0x020617);
          viewport.scene.fog = new THREE.FogExp2(0x020617, 0.00055);

          viewport.camera = new THREE.PerspectiveCamera(50, width / height, 1, 5000);
          viewport.renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'default' });
          viewport.renderer.setSize(width, height);
          viewport.renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2));
          if (els.viewportContainer) {
            els.viewportContainer.appendChild(viewport.renderer.domElement);
          }

          const ambient = new THREE.AmbientLight(0xe2e8f0, 0.9);
          viewport.scene.add(ambient);

          const dirLight = new THREE.DirectionalLight(0xffffff, 1.0);
          dirLight.position.set(160, 260, 190);
          viewport.scene.add(dirLight);

          const sunPointLight = new THREE.PointLight(0xfff3c4, 1.9, 1800, 0.5);
          sunPointLight.position.set(0, 0, 0);
          viewport.scene.add(sunPointLight);
          viewport.sunLight = sunPointLight;

          const starCount = 1500;
          const starGeo = new THREE.BufferGeometry();
          const starPositions = new Float32Array(starCount * 3);
          for (let i = 0; i < starCount; i++) {
            const r = 450 + Math.random() * 1650;
            const theta = Math.random() * Math.PI * 2;
            const phi = Math.acos(2 * Math.random() - 1);
            starPositions[i * 3] = r * Math.sin(phi) * Math.cos(theta);
            starPositions[i * 3 + 1] = r * Math.cos(phi);
            starPositions[i * 3 + 2] = r * Math.sin(phi) * Math.sin(theta);
          }
          starGeo.setAttribute('position', new THREE.BufferAttribute(starPositions, 3));
          const starMat = new THREE.PointsMaterial({
            color: 0xe2e8f0,
            size: 2.2,
            transparent: true,
            opacity: 0.75
          });
          viewport.starfield = new THREE.Points(starGeo, starMat);
          viewport.scene.add(viewport.starfield);
          return true;
        } catch (err) {
          viewport.renderer = null;
          viewport.scene = null;
          viewport.camera = null;
          return false;
        }
      }

      function createCelestialTexture(kind, baseHex, id) {
        if (!isThreeAvailable()) return null;
        try {
          const canvas = document.createElement('canvas');
          canvas.width = 512;
          canvas.height = 256;
          const ctx = canvas.getContext('2d');
          const c = new THREE.Color(baseHex || '#38bdf8');
          const rBase = Math.round(c.r * 255);
          const gBase = Math.round(c.g * 255);
          const bBase = Math.round(c.b * 255);

          ctx.fillStyle = `rgb(${rBase}, ${gBase}, ${bBase})`;
          ctx.fillRect(0, 0, canvas.width, canvas.height);

          let seed = 1337;
          for (let i = 0; i < String(id || '').length; i++) {
            seed = (seed * 31 + String(id).charCodeAt(i)) % 2147483647;
          }
          const rand = () => {
            seed = (seed * 16807) % 2147483647;
            return (seed - 1) / 2147483646;
          };

          if (kind === 'star') {
            for (let i = 0; i < 420; i++) {
              const x = rand() * canvas.width;
              const y = rand() * canvas.height;
              const rad = 2 + rand() * 11;
              const bright = rand() > 0.35;
              ctx.fillStyle = bright ? 'rgba(255, 251, 235, 0.22)' : 'rgba(180, 83, 9, 0.24)';
              ctx.beginPath();
              ctx.arc(x, y, rad, 0, Math.PI * 2);
              ctx.fill();
            }
          } else if (id === 'planet-jupiter' || id === 'planet-saturn') {
            const bands = id === 'planet-jupiter' ? 18 : 12;
            for (let i = 0; i < bands; i++) {
              const y = (i / bands) * canvas.height;
              const h = (canvas.height / bands) * (0.65 + rand() * 0.5);
              const shift = (rand() - 0.5) * 65;
              ctx.fillStyle = `rgba(${Math.min(255, Math.max(0, rBase + shift))}, ${Math.min(255, Math.max(0, gBase + shift))}, ${Math.min(255, Math.max(0, bBase + shift))}, 0.55)`;
              ctx.fillRect(0, y, canvas.width, h);
            }
            if (id === 'planet-jupiter') {
              ctx.fillStyle = 'rgba(194, 65, 12, 0.72)';
              ctx.beginPath();
              ctx.ellipse(canvas.width * 0.62, canvas.height * 0.64, 34, 18, 0.06, 0, Math.PI * 2);
              ctx.fill();
            }
          } else if (id === 'planet-earth') {
            ctx.fillStyle = 'rgba(34, 197, 94, 0.65)';
            for (let i = 0; i < 22; i++) {
              ctx.beginPath();
              ctx.ellipse(rand() * canvas.width, 36 + rand() * (canvas.height - 72), 18 + rand() * 44, 10 + rand() * 24, rand() * Math.PI, 0, Math.PI * 2);
              ctx.fill();
            }
            ctx.fillStyle = 'rgba(248, 250, 252, 0.38)';
            for (let i = 0; i < 36; i++) {
              ctx.beginPath();
              ctx.ellipse(rand() * canvas.width, rand() * canvas.height, 22 + rand() * 45, 4 + rand() * 9, (rand() - 0.5) * 0.25, 0, Math.PI * 2);
              ctx.fill();
            }
          } else {
            for (let i = 0; i < 140; i++) {
              const x = rand() * canvas.width;
              const y = rand() * canvas.height;
              const rad = 2 + rand() * 14;
              ctx.fillStyle = rand() > 0.5 ? 'rgba(255, 255, 255, 0.12)' : 'rgba(15, 23, 42, 0.22)';
              ctx.beginPath();
              ctx.arc(x, y, rad, 0, Math.PI * 2);
              ctx.fill();
            }
          }

          const tex = new THREE.CanvasTexture(canvas);
          tex.wrapS = THREE.RepeatWrapping;
          tex.wrapT = THREE.ClampToEdgeWrapping;
          tex.needsUpdate = true;
          return tex;
        } catch (err) {
          return null;
        }
      }

      function createSaturnRingTexture() {
        if (!isThreeAvailable()) return null;
        try {
          const canvas = document.createElement('canvas');
          canvas.width = 256;
          canvas.height = 32;
          const ctx = canvas.getContext('2d');
          for (let x = 0; x < canvas.width; x++) {
            const t = x / canvas.width;
            const inCassiniGap = t > 0.64 && t < 0.72;
            const alpha = inCassiniGap ? 0.12 : 0.45 + 0.4 * Math.sin(t * Math.PI * 10);
            ctx.fillStyle = `rgba(254, 240, 138, ${Math.max(0.08, Math.min(0.95, alpha))})`;
            ctx.fillRect(x, 0, 1, canvas.height);
          }
          const tex = new THREE.CanvasTexture(canvas);
          tex.needsUpdate = true;
          return tex;
        } catch (err) {
          return null;
        }
      }

      function rebuild3DScene() {
        if (!isThreeAvailable() || !viewport.scene) return;

        const disposeObject3D = (obj) => {
          if (!obj) return;
          obj.traverse((child) => {
            if (child.geometry && typeof child.geometry.dispose === 'function') {
              child.geometry.dispose();
            }
            const mats = Array.isArray(child.material) ? child.material : [child.material];
            mats.forEach((mat) => {
              if (!mat) return;
              if (mat.map && typeof mat.map.dispose === 'function') {
                mat.map.dispose();
              }
              if (typeof mat.dispose === 'function') {
                mat.dispose();
              }
            });
          });
        };

        viewport.starMeshes.forEach((entry) => {
          if (entry && entry.group) {
            viewport.scene.remove(entry.group);
            disposeObject3D(entry.group);
          }
        });
        viewport.starMeshes.clear();

        viewport.planetMeshes.forEach((entry) => {
          if (!entry) return;
          if (entry.group) {
            viewport.scene.remove(entry.group);
            disposeObject3D(entry.group);
          }
          if (entry.orbitLine) {
            viewport.scene.remove(entry.orbitLine);
            disposeObject3D(entry.orbitLine);
          }
        });
        viewport.planetMeshes.clear();

        appState.data.stars.forEach((star) => {
          const group = new THREE.Group();
          const r = Number(star.radius || 12);
          const colorHex = new THREE.Color(star.color || '#f59e0b');
          const sunTex = createCelestialTexture('star', star.color || '#fbbf24', star.id);

          const coreGeo = new THREE.SphereGeometry(r, 48, 48);
          const coreMat = new THREE.MeshBasicMaterial({
            map: sunTex || null,
            color: sunTex ? 0xffffff : colorHex
          });
          const core = new THREE.Mesh(coreGeo, coreMat);
          core.userData = { kind: 'star', id: star.id };
          group.add(core);

          const haloGeo = new THREE.SphereGeometry(r * 1.42, 32, 32);
          const haloMat = new THREE.MeshBasicMaterial({
            color: colorHex,
            transparent: true,
            opacity: 0.22,
            blending: THREE.AdditiveBlending,
            depthWrite: false
          });
          const halo = new THREE.Mesh(haloGeo, haloMat);
          group.add(halo);

          const ringGeo = new THREE.RingGeometry(r * 1.55, r * 1.72, 40);
          const ringMat = new THREE.MeshBasicMaterial({
            color: 0xfef08a,
            side: THREE.DoubleSide,
            transparent: true,
            opacity: 0.85
          });
          const ring = new THREE.Mesh(ringGeo, ringMat);
          ring.rotation.x = Math.PI / 2;
          ring.visible = appState.data.selectedKind === 'star' && appState.data.selectedId === star.id;
          group.add(ring);

          const pos = star.position || { x: 0, y: 0, z: 0 };
          group.position.set(pos.x || 0, pos.y || 0, pos.z || 0);
          viewport.scene.add(group);
          viewport.starMeshes.set(star.id, { group, core, halo, ring });
        });

        appState.data.planets.forEach((planet) => {
          const star = getStarById(planet.starId) || appState.data.stars[0];
          const center = star && star.position ? star.position : { x: 0, y: 0, z: 0 };
          const r = Number(planet.radius || 3.8);
          const orbitR = Number(planet.orbitRadius || 40);
          const colorHex = new THREE.Color(planet.color || '#38bdf8');
          const planetTex = createCelestialTexture('planet', planet.color || '#38bdf8', planet.id);

          const group = new THREE.Group();
          const bodyGeo = new THREE.SphereGeometry(r, 36, 36);
          const bodyMat = new THREE.MeshStandardMaterial({
            map: planetTex || null,
            color: planetTex ? 0xffffff : colorHex,
            roughness: ['planet-earth', 'planet-neptune', 'planet-uranus'].includes(planet.id) ? 0.52 : 0.78,
            metalness: 0.06
          });
          const body = new THREE.Mesh(bodyGeo, bodyMat);
          body.userData = { kind: 'planet', id: planet.id };
          group.add(body);

          if (['planet-earth', 'planet-venus', 'planet-jupiter', 'planet-saturn', 'planet-uranus', 'planet-neptune'].includes(planet.id)) {
            const atmoGeo = new THREE.SphereGeometry(r * 1.045, 32, 32);
            const atmoMat = new THREE.MeshBasicMaterial({
              color: colorHex,
              transparent: true,
              opacity: planet.id === 'planet-earth' ? 0.16 : 0.1,
              blending: THREE.AdditiveBlending,
              depthWrite: false
            });
            group.add(new THREE.Mesh(atmoGeo, atmoMat));
          }

          const selGeo = new THREE.RingGeometry(r * 1.35, r * 1.52, 28);
          const selMat = new THREE.MeshBasicMaterial({
            color: 0xf8fafc,
            side: THREE.DoubleSide,
            transparent: true,
            opacity: 0.9
          });
          const selRing = new THREE.Mesh(selGeo, selMat);
          selRing.rotation.x = Math.PI / 2;
          selRing.visible = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;
          group.add(selRing);

          if (planet.id === 'planet-saturn') {
            const ringTex = createSaturnRingTexture();
            const saturnRingGeo = new THREE.RingGeometry(r * 1.28, r * 2.28, 64);
            const saturnRingMat = new THREE.MeshBasicMaterial({
              map: ringTex || null,
              color: ringTex ? 0xffffff : 0xfef08a,
              side: THREE.DoubleSide,
              transparent: true,
              opacity: 0.85,
              depthWrite: false
            });
            const saturnRing = new THREE.Mesh(saturnRingGeo, saturnRingMat);
            saturnRing.rotation.x = Math.PI * 0.42;
            saturnRing.rotation.y = 0.22;
            group.add(saturnRing);
          }

          const segments = 96;
          const pts = [];
          const inc = Number(planet.inclination || 0);
          for (let i = 0; i <= segments; i++) {
            const a = (i / segments) * Math.PI * 2;
            pts.push(
              new THREE.Vector3(
                (center.x || 0) + Math.cos(a) * orbitR,
                (center.y || 0) + Math.sin(a * 1.5) * Math.sin(inc) * orbitR * 0.45,
                (center.z || 0) + Math.sin(a) * orbitR
              )
            );
          }
          const orbitGeo = new THREE.BufferGeometry().setFromPoints(pts);
          const isSelected = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;
          const orbitMat = new THREE.LineBasicMaterial({
            color: isSelected ? 0x38bdf8 : 0x475569,
            transparent: true,
            opacity: isSelected ? 0.78 : 0.34
          });
          const orbitLine = new THREE.Line(orbitGeo, orbitMat);
          viewport.scene.add(orbitLine);
          viewport.scene.add(group);

          viewport.planetMeshes.set(planet.id, { group, body, selRing, orbitLine });
        });
      }

      function clampVal(v, min, max) {
        return Math.min(max, Math.max(min, v));
      }

      function update3DTransforms(dt) {
        if (!isThreeAvailable() || !viewport.camera) return;

        if (!appState.data.orbitPaused) {
          viewport.elapsedOrbitTime += dt;
        }

        appState.data.stars.forEach((star, idx) => {
          const entry = viewport.starMeshes.get(star.id);
          if (!entry) return;
          const pos = star.position || { x: 0, y: 0, z: 0 };
          entry.group.position.set(pos.x || 0, pos.y || 0, pos.z || 0);
          if (idx === 0 && viewport.sunLight) {
            viewport.sunLight.position.set(pos.x || 0, pos.y || 0, pos.z || 0);
          }
          entry.core.rotation.y += dt * 0.22;
          entry.halo.rotation.y -= dt * 0.12;
          const isSel = appState.data.selectedKind === 'star' && appState.data.selectedId === star.id;
          entry.ring.visible = isSel;
          if (isSel) entry.ring.rotation.z += dt * 0.8;
        });

        appState.data.planets.forEach((planet) => {
          const entry = viewport.planetMeshes.get(planet.id);
          if (!entry) return;
          const pos = getPlanetWorldPosition(planet, viewport.elapsedOrbitTime);
          entry.group.position.set(pos.x, pos.y, pos.z);
          entry.body.rotation.y += dt * 0.9;
          const isSel = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;
          entry.selRing.visible = isSel;
          if (isSel) entry.selRing.rotation.z += dt * 1.2;
        });

        if (appState.data.lockedTarget) {
          if (appState.data.lockedTarget.kind === 'star') {
            const s = getStarById(appState.data.lockedTarget.id);
            if (s && s.position) {
              viewport.cameraTarget.x += ((s.position.x || 0) - viewport.cameraTarget.x) * 0.14;
              viewport.cameraTarget.y += ((s.position.y || 0) - viewport.cameraTarget.y) * 0.14;
              viewport.cameraTarget.z += ((s.position.z || 0) - viewport.cameraTarget.z) * 0.14;
            }
          } else if (appState.data.lockedTarget.kind === 'planet') {
            const p = getPlanetById(appState.data.lockedTarget.id);
            if (p) {
              const pos = getPlanetWorldPosition(p, viewport.elapsedOrbitTime);
              viewport.cameraTarget.x += (pos.x - viewport.cameraTarget.x) * 0.16;
              viewport.cameraTarget.y += (pos.y - viewport.cameraTarget.y) * 0.16;
              viewport.cameraTarget.z += (pos.z - viewport.cameraTarget.z) * 0.16;
            }
          }
        }

        const r = clampVal(Number(viewport.cameraSpherical.radius) || BASE_DISTANCE, 25, 1800);
        const phi = clampVal(Number(viewport.cameraSpherical.phi) || Math.PI * 0.36, 0.12, Math.PI - 0.12);
        const theta = Number(viewport.cameraSpherical.theta) || 0.25;

        const sinPhi = Math.sin(phi);
        viewport.camera.position.set(
          viewport.cameraTarget.x + r * sinPhi * Math.sin(theta),
          viewport.cameraTarget.y + r * Math.cos(phi),
          viewport.cameraTarget.z + r * sinPhi * Math.cos(theta)
        );
        viewport.camera.lookAt(viewport.cameraTarget.x, viewport.cameraTarget.y, viewport.cameraTarget.z);

        if (viewport.starfield) {
          viewport.starfield.rotation.y += dt * 0.012;
        }
      }

      function render2DCanvas() {
        const ctx = viewport.ctx2D;
        const canvas = viewport.canvas2D;
        if (!ctx || !canvas) return;

        const w = canvas.width;
        const h = canvas.height;
        const zoomScale = BASE_DISTANCE / Math.max(25, Number(viewport.cameraSpherical.radius) || BASE_DISTANCE);

        ctx.clearRect(0, 0, w, h);
        ctx.fillStyle = '#020617';
        ctx.fillRect(0, 0, w, h);

        ctx.strokeStyle = 'rgba(51, 65, 85, 0.24)';
        ctx.lineWidth = 1;
        const gridStep = Math.max(28, 60 * zoomScale);
        const offsetX = (((w * 0.5 + viewport.pan2D.x) % gridStep) + gridStep) % gridStep;
        const offsetY = (((h * 0.5 + viewport.pan2D.y) % gridStep) + gridStep) % gridStep;
        for (let gx = offsetX; gx < w; gx += gridStep) {
          ctx.beginPath();
          ctx.moveTo(gx, 0);
          ctx.lineTo(gx, h);
          ctx.stroke();
        }
        for (let gy = offsetY; gy < h; gy += gridStep) {
          ctx.beginPath();
          ctx.moveTo(0, gy);
          ctx.lineTo(w, gy);
          ctx.stroke();
        }

        if (appState.data.lockedTarget) {
          if (appState.data.lockedTarget.kind === 'star') {
            const s = getStarById(appState.data.lockedTarget.id);
            if (s && s.position) {
              viewport.pan2D.x = -(s.position.x || 0) * zoomScale;
              viewport.pan2D.y = -(s.position.z || 0) * zoomScale;
            }
          } else if (appState.data.lockedTarget.kind === 'planet') {
            const p = getPlanetById(appState.data.lockedTarget.id);
            if (p) {
              const pos = getPlanetWorldPosition(p, viewport.elapsedOrbitTime);
              viewport.pan2D.x = -pos.x * zoomScale;
              viewport.pan2D.y = -pos.z * zoomScale;
            }
          }
        }

        const toScreen = (wx, wz) => ({
          x: w * 0.5 + viewport.pan2D.x + wx * zoomScale,
          y: h * 0.5 + viewport.pan2D.y + wz * zoomScale
        });

        viewport.projectedHitTargets = [];

        appState.data.planets.forEach((planet) => {
          const star = getStarById(planet.starId) || appState.data.stars[0];
          const centerPos = star && star.position ? star.position : { x: 0, y: 0, z: 0 };
          const centerScreen = toScreen(centerPos.x || 0, centerPos.z || 0);
          const orbitPx = Number(planet.orbitRadius || 40) * zoomScale;
          const isSel = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;

          ctx.beginPath();
          ctx.arc(centerScreen.x, centerScreen.y, orbitPx, 0, Math.PI * 2);
          ctx.strokeStyle = isSel ? 'rgba(56, 189, 248, 0.72)' : 'rgba(100, 116, 139, 0.28)';
          ctx.lineWidth = isSel ? 1.8 : 1;
          ctx.stroke();
        });

        appState.data.stars.forEach((star) => {
          const pos = star.position || { x: 0, y: 0, z: 0 };
          const sc = toScreen(pos.x || 0, pos.z || 0);
          const rPx = Math.max(12, Number(star.radius || 14) * zoomScale * 0.95);
          const isSel = appState.data.selectedKind === 'star' && appState.data.selectedId === star.id;

          const grad = ctx.createRadialGradient(sc.x, sc.y, rPx * 0.15, sc.x, sc.y, rPx * 2.25);
          grad.addColorStop(0, star.color || '#f59e0b');
          grad.addColorStop(0.55, 'rgba(251, 191, 36, 0.25)');
          grad.addColorStop(1, 'rgba(245, 158, 11, 0)');
          ctx.fillStyle = grad;
          ctx.beginPath();
          ctx.arc(sc.x, sc.y, rPx * 2.25, 0, Math.PI * 2);
          ctx.fill();

          const sunSphereGrad = ctx.createRadialGradient(sc.x - rPx * 0.25, sc.y - rPx * 0.25, rPx * 0.1, sc.x, sc.y, rPx);
          sunSphereGrad.addColorStop(0, '#fffbeb');
          sunSphereGrad.addColorStop(0.55, star.color || '#fbbf24');
          sunSphereGrad.addColorStop(1, '#d97706');
          ctx.fillStyle = sunSphereGrad;
          ctx.beginPath();
          ctx.arc(sc.x, sc.y, rPx, 0, Math.PI * 2);
          ctx.fill();

          if (isSel) {
            ctx.strokeStyle = '#38bdf8';
            ctx.lineWidth = 2.5;
            ctx.beginPath();
            ctx.arc(sc.x, sc.y, rPx + 6, 0, Math.PI * 2);
            ctx.stroke();
          }

          ctx.fillStyle = '#f8fafc';
          ctx.font = '600 12px Inter, sans-serif';
          ctx.textAlign = 'center';
          ctx.fillText(star.name, sc.x, sc.y + rPx + 16);

          viewport.projectedHitTargets.push({
            kind: 'star',
            id: star.id,
            x: sc.x,
            y: sc.y,
            hitRadius: Math.max(16, rPx + 6)
          });
        });

        appState.data.planets.forEach((planet) => {
          const pos = getPlanetWorldPosition(planet, viewport.elapsedOrbitTime);
          const sc = toScreen(pos.x, pos.z);
          const rPx = Math.max(5, Number(planet.radius || 3.8) * zoomScale * 1.15);
          const isSel = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;

          if (planet.id === 'planet-saturn') {
            ctx.save();
            ctx.translate(sc.x, sc.y);
            ctx.rotate(-0.28);
            ctx.strokeStyle = 'rgba(254, 240, 138, 0.72)';
            ctx.lineWidth = Math.max(2, rPx * 0.32);
            ctx.beginPath();
            ctx.ellipse(0, 0, rPx * 2.05, rPx * 0.62, 0, 0, Math.PI * 2);
            ctx.stroke();
            ctx.restore();
          }

          const planetSphereGrad = ctx.createRadialGradient(
            sc.x - rPx * 0.35,
            sc.y - rPx * 0.35,
            rPx * 0.1,
            sc.x,
            sc.y,
            rPx
          );
          planetSphereGrad.addColorStop(0, '#f8fafc');
          planetSphereGrad.addColorStop(0.22, planet.color || '#38bdf8');
          planetSphereGrad.addColorStop(1, '#090d16');
          ctx.fillStyle = planetSphereGrad;
          ctx.beginPath();
          ctx.arc(sc.x, sc.y, rPx, 0, Math.PI * 2);
          ctx.fill();

          if (isSel) {
            ctx.strokeStyle = '#38bdf8';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.arc(sc.x, sc.y, rPx + 4.5, 0, Math.PI * 2);
            ctx.stroke();
          }

          ctx.fillStyle = '#e2e8f0';
          ctx.font = '500 11px Inter, sans-serif';
          ctx.textAlign = 'center';
          ctx.fillText(planet.name, sc.x, sc.y + rPx + 14);

          viewport.projectedHitTargets.push({
            kind: 'planet',
            id: planet.id,
            x: sc.x,
            y: sc.y,
            hitRadius: Math.max(12, rPx + 5)
          });
        });
      }

      function renderLoop(now) {
        try {
          const currentNow = typeof now === 'number' ? now : performance.now();
          const dt = Math.min(0.05, Math.max(0.001, (currentNow - viewport.lastFrameTime) / 1000));
          viewport.lastFrameTime = currentNow;

          if (appState.data.viewMode === '3d' && viewport.renderer && viewport.scene && viewport.camera) {
            update3DTransforms(dt);
            viewport.renderer.render(viewport.scene, viewport.camera);
          } else {
            if (!appState.data.orbitPaused) {
              viewport.elapsedOrbitTime += dt;
            }
            render2DCanvas();
          }
        } catch (err) {
          try {
            if (!appState.data.orbitPaused) {
              viewport.elapsedOrbitTime += 0.016;
            }
            render2DCanvas();
          } catch (fallbackErr) {
            // 最小フォールバック
          }
        }
        requestAnimationFrame(renderLoop);
      }

      function startRenderLoop() {
        if (viewport.loopStarted) return;
        viewport.loopStarted = true;
        viewport.lastFrameTime = performance.now();
        requestAnimationFrame(renderLoop);
      }

      function handleViewportResize() {
        const w = Math.max(320, (els.viewportContainer && els.viewportContainer.clientWidth) || window.innerWidth || 1024);
        const h = Math.max(240, (els.viewportContainer && els.viewportContainer.clientHeight) || window.innerHeight || 768);
        if (viewport.camera && viewport.renderer) {
          viewport.camera.aspect = w / h;
          viewport.camera.updateProjectionMatrix();
          viewport.renderer.setSize(w, h);
        }
        if (viewport.canvas2D) {
          viewport.canvas2D.width = w;
          viewport.canvas2D.height = h;
        }
      }

      function setCameraDistance(dist, persist = true) {
        const clamped = Math.min(1800, Math.max(25, Number(dist) || BASE_DISTANCE));
        viewport.cameraSpherical.radius = clamped;
        appState.data.cameraDistance = clamped;
        syncZoomUI();
        if (persist) saveStateToLocalStorage(false);
      }

      function syncZoomUI() {
        const r = Number(viewport.cameraSpherical.radius) || BASE_DISTANCE;
        if (els.cameraZoomRange) els.cameraZoomRange.value = String(Math.round(r));
        const pct = Math.max(14, Math.min(1700, Math.round((BASE_DISTANCE / r) * 100)));
        if (els.cameraZoomPercent) els.cameraZoomPercent.textContent = `${pct}%`;
      }

      function updateLockStatusUI() {
        const lock = appState.data.lockedTarget;
        if (!lock) {
          if (els.lockStatusText) els.lockStatusText.textContent = '自由視点';
          if (els.btnUnlockCamera) els.btnUnlockCamera.classList.add('hidden');
          return;
        }
        if (lock.kind === 'star') {
          const s = getStarById(lock.id);
          if (els.lockStatusText) els.lockStatusText.textContent = s ? `恒星ロック: ${s.name}` : '自由視点';
        } else {
          const p = getPlanetById(lock.id);
          if (els.lockStatusText) els.lockStatusText.textContent = p ? `惑星ロック: ${p.name}` : '自由視点';
        }
        if (els.btnUnlockCamera) els.btnUnlockCamera.classList.remove('hidden');
      }

      function selectEntity(kind, id, lockCamera = false, openInspector = false) {
        appState.data.selectedKind = kind;
        appState.data.selectedId = id;
        if (lockCamera) {
          appState.data.lockedTarget = { kind, id };
        }
        if (openInspector && els.inspectorPanel) {
          els.inspectorPanel.classList.remove('hidden');
        }
        rebuild3DScene();
        renderAllUI();
        saveStateToLocalStorage(false);
      }

      function openPlanetContent(planet) {
        if (!planet) return;
        const pType = normalizePlanetType(planet);
        if (pType === 'image') {
          openImageViewer(planet.id);
        } else {
          openDocEditor(planet.id);
        }
      }

      function hitTestAtClientPos(clientX, clientY) {
        if (!els.viewportContainer) return null;
        const rect = els.viewportContainer.getBoundingClientRect();
        const x = clientX - rect.left;
        const y = clientY - rect.top;

        if (appState.data.viewMode === '2d' || !viewport.camera || !viewport.raycaster) {
          for (let i = viewport.projectedHitTargets.length - 1; i >= 0; i--) {
            const t = viewport.projectedHitTargets[i];
            const dist = Math.hypot(x - t.x, y - t.y);
            if (dist <= t.hitRadius) return { kind: t.kind, id: t.id };
          }
          return null;
        }

        viewport.pointerNDC.x = (x / Math.max(1, rect.width)) * 2 - 1;
        viewport.pointerNDC.y = -(y / Math.max(1, rect.height)) * 2 + 1;
        viewport.raycaster.setFromCamera(viewport.pointerNDC, viewport.camera);

        const candidates = [];
        viewport.planetMeshes.forEach((entry) => candidates.push(entry.body));
        viewport.starMeshes.forEach((entry) => candidates.push(entry.core));

        const hits = viewport.raycaster.intersectObjects(candidates, false);
        if (hits.length > 0 && hits[0].object && hits[0].object.userData) {
          return { kind: hits[0].object.userData.kind, id: hits[0].object.userData.id };
        }
        return null;
      }

      function bindViewportEvents() {
        const container = els.viewportContainer;
        if (!container) return;

        onEl(container, 'pointerdown', (e) => {
          if (e.button === 2) {
            viewport.dragMode = 'pan';
          } else {
            viewport.dragMode = appState.data.viewMode === '2d' ? 'pan' : 'orbit';
          }
          viewport.isDragging = true;
          viewport.dragStart = { x: e.clientX, y: e.clientY };
          viewport.pointerDownPos = { x: e.clientX, y: e.clientY };
        });

        window.addEventListener('pointermove', (e) => {
          if (!viewport.isDragging) return;
          const dx = e.clientX - viewport.dragStart.x;
          const dy = e.clientY - viewport.dragStart.y;
          viewport.dragStart = { x: e.clientX, y: e.clientY };

          if (appState.data.viewMode === '2d') {
            appState.data.lockedTarget = null;
            updateLockStatusUI();
            viewport.pan2D.x += dx;
            viewport.pan2D.y += dy;
          } else if (viewport.dragMode === 'orbit') {
            viewport.cameraSpherical.theta -= dx * 0.0065;
            viewport.cameraSpherical.phi = clampVal(viewport.cameraSpherical.phi - dy * 0.0065, 0.12, Math.PI - 0.12);
          } else if (viewport.dragMode === 'pan') {
            appState.data.lockedTarget = null;
            updateLockStatusUI();
            const panScale = viewport.cameraSpherical.radius * 0.0016;
            viewport.cameraTarget.x -= dx * panScale;
            viewport.cameraTarget.y += dy * panScale;
          }
        });

        window.addEventListener('pointerup', (e) => {
          if (!viewport.isDragging) return;
          const moved = Math.hypot(e.clientX - viewport.pointerDownPos.x, e.clientY - viewport.pointerDownPos.y);
          viewport.isDragging = false;
          viewport.dragMode = null;

          if (moved < 6) {
            const hit = hitTestAtClientPos(e.clientX, e.clientY);
            if (hit) {
              selectEntity(hit.kind, hit.id, true, false);
            }
          }
        });

        onEl(container, 'dblclick', (e) => {
          const hit = hitTestAtClientPos(e.clientX, e.clientY);
          if (hit && hit.kind === 'planet') {
            const planet = getPlanetById(hit.id);
            if (planet) openPlanetContent(planet);
          } else if (hit && hit.kind === 'star') {
            selectEntity('star', hit.id, true, true);
          }
        });

        onEl(
          container,
          'wheel',
          (e) => {
            e.preventDefault();
            const factor = e.deltaY > 0 ? 1.1 : 0.9;
            setCameraDistance(viewport.cameraSpherical.radius * factor, true);
          },
          { passive: false }
        );

        onEl(container, 'contextmenu', (e) => e.preventDefault());
      }

      // 7. 星系エクスプローラー描画 & ドラッグ移籍 & 恒星タブ開閉
      // =========================================================================
      function getTypeBadge(type) {
        if (type === 'image') {
          return { icon: 'ph-image', color: 'text-pink-400', label: '画像' };
        }
        return { icon: 'ph-file-text', color: 'text-sky-400', label: '文書' };
      }

      function matchesFilters(planet) {
        const normalizedType = normalizePlanetType(planet);
        if (appState.data.typeFilter !== 'all' && normalizedType !== appState.data.typeFilter) {
          return false;
        }
        if (appState.searchQuery.trim() !== '') {
          const q = appState.searchQuery.trim().toLowerCase();
          return String(planet.name || '').toLowerCase().includes(q);
        }
        return true;
      }

      function toggleStarCollapsed(starId) {
        if (!Array.isArray(appState.data.collapsedStarIds)) {
          appState.data.collapsedStarIds = [];
        }
        const idx = appState.data.collapsedStarIds.indexOf(starId);
        if (idx >= 0) {
          appState.data.collapsedStarIds.splice(idx, 1);
        } else {
          appState.data.collapsedStarIds.push(starId);
        }
        renderExplorer();
        saveStateToLocalStorage(false);
      }

      function togglePlanetExpanded(planetId) {
        const idx = appState.data.expandedPlanetIds.indexOf(planetId);
        if (idx >= 0) {
          appState.data.expandedPlanetIds.splice(idx, 1);
        } else {
          appState.data.expandedPlanetIds.push(planetId);
        }
        renderExplorer();
        saveStateToLocalStorage(false);
      }

      function renderExplorer() {
        const listEl = els.explorerList;
        if (!listEl) return;
        listEl.innerHTML = '';

        const totalStars = appState.data.stars.length;
        const totalPlanets = appState.data.planets.length;
        if (els.explorerCounts) {
          els.explorerCounts.textContent = `${totalStars}恒星 / ${totalPlanets}惑星`;
        }

        if (els.tabStarTree && els.tabAllFiles) {
          if (appState.data.explorerTab === 'tree') {
            els.tabStarTree.className = 'rounded-lg py-1 font-medium bg-sky-500/25 text-sky-200 transition';
            els.tabAllFiles.className = 'rounded-lg py-1 font-medium text-slate-400 hover:text-slate-200 transition';
          } else {
            els.tabAllFiles.className = 'rounded-lg py-1 font-medium bg-sky-500/25 text-sky-200 transition';
            els.tabStarTree.className = 'rounded-lg py-1 font-medium text-slate-400 hover:text-slate-200 transition';
          }
        }

        els.typeFilterBtns.forEach((btn) => {
          const active = btn.dataset.typeFilter === appState.data.typeFilter;
          btn.className = active
            ? 'type-filter-btn shrink-0 rounded-lg px-2 py-0.5 bg-slate-800 text-sky-300 border border-sky-500/40'
            : 'type-filter-btn shrink-0 rounded-lg px-2 py-0.5 bg-slate-900 text-slate-400 hover:text-slate-200 border border-slate-800';
        });

        if (appState.data.explorerTab === 'tree') {
          const collapsedSet = new Set(appState.data.collapsedStarIds || []);
          appState.data.stars.forEach((star) => {
            const isCollapsed = collapsedSet.has(star.id);
            const starCard = document.createElement('div');
            const isStarSelected = appState.data.selectedKind === 'star' && appState.data.selectedId === star.id;
            starCard.className = `rounded-xl border transition ${
              isStarSelected ? 'bg-slate-800/90 border-amber-400/50' : 'bg-slate-950/65 border-slate-800/90 hover:border-slate-700'
            }`;

            starCard.addEventListener('dragover', (e) => {
              e.preventDefault();
              starCard.classList.add('border-sky-400', 'bg-sky-500/10');
            });
            starCard.addEventListener('dragleave', () => {
              starCard.classList.remove('border-sky-400', 'bg-sky-500/10');
            });
            starCard.addEventListener('drop', (e) => {
              e.preventDefault();
              starCard.classList.remove('border-sky-400', 'bg-sky-500/10');
              const pid = (e.dataTransfer && e.dataTransfer.getData('text/plain')) || appState.draggedPlanetId;
              const planet = getPlanetById(pid);
              if (planet && planet.starId !== star.id) {
                planet.starId = star.id;
                if (collapsedSet.has(star.id)) {
                  toggleStarCollapsed(star.id);
                }
                rebuild3DScene();
                renderAllUI();
                saveStateToLocalStorage(true);
                showToast(`「${planet.name}」を ${star.name} へ移籍しました`, 'ph-arrows-left-right');
              }
              appState.draggedPlanetId = null;
            });

            const planets = getPlanetsForStar(star.id).filter(matchesFilters);

            const starHeader = document.createElement('div');
            starHeader.className = 'flex items-center justify-between px-2.5 py-2 cursor-pointer select-none';
            starHeader.innerHTML = `
              <div class="flex items-center gap-1.5 min-w-0 flex-1">
                <button type="button" title="${isCollapsed ? '恒星タブを展開' : '恒星タブを折りたたむ'}" aria-expanded="${!isCollapsed}" class="toggle-star-btn rounded p-0.5 text-slate-400 hover:text-amber-200 hover:bg-slate-800/80 transition shrink-0">
                  <i class="ph ${isCollapsed ? 'ph-caret-right' : 'ph-caret-down'} text-xs"></i>
                </button>
                <i class="ph-fill ph-sun text-sm shrink-0" style="color:${escapeHtml(star.color || '#f59e0b')}"></i>
                <span class="font-semibold text-slate-100 truncate">${escapeHtml(star.name)}</span>
              </div>
              <span class="text-[10px] font-mono text-slate-400 shrink-0 ml-1">${planets.length}</span>
            `;

            const toggleStarBtn = starHeader.querySelector('.toggle-star-btn');
            if (toggleStarBtn) {
              toggleStarBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                toggleStarCollapsed(star.id);
              });
            }

            starHeader.addEventListener('click', () => {
              selectEntity('star', star.id, true, true);
            });
            starCard.appendChild(starHeader);

            const planetsContainer = document.createElement('div');
            planetsContainer.className = `px-2 pb-2 space-y-1 ${isCollapsed ? 'hidden' : ''}`;

            if (!isCollapsed && planets.length === 0) {
              const emptyHint = document.createElement('div');
              emptyHint.className = 'px-2 py-1.5 text-[10px] text-slate-500';
              emptyHint.textContent = '表示対象の惑星はありません';
              planetsContainer.appendChild(emptyHint);
            }

            planets.forEach((planet) => {
              const isExpanded = appState.data.expandedPlanetIds.includes(planet.id);
              const isSelected = appState.data.selectedKind === 'planet' && appState.data.selectedId === planet.id;
              const badge = getTypeBadge(normalizePlanetType(planet));

              const pRow = document.createElement('div');

      function getSolarSystemOuterBoundary() {
        const sun = getStarById('star-sun') || appState.data.stars[0];
        const sunPlanets = sun ? getPlanetsForStar(sun.id) : [];
        let maxOuter = 225;
        sunPlanets.forEach((p) => {
          const outer = Number(p.orbitRadius || 0) + Number(p.radius || 0) + 42;
          if (outer > maxOuter) maxOuter = outer;
        });
        return maxOuter;
      }

      function enforceStarExclusionRule(rawPos, ignoreStarId = null, notifyIfAdjusted = false) {
        const sun = getStarById('star-sun') || appState.data.stars[0];
        const sunPos = sun && sun.position ? sun.position : { x: 0, y: 0, z: 0 };
        const minSolarDist = getSolarSystemOuterBoundary();

        let x = Number(rawPos && rawPos.x) || 0;
        let y = Number(rawPos && rawPos.y) || 0;
        let z = Number(rawPos && rawPos.z) || 0;
        let adjusted = false;

        if (!sun || sun.id !== ignoreStarId) {
          const dxSun = x - (sunPos.x || 0);
          const dzSun = z - (sunPos.z || 0);
          const distSun = Math.hypot(dxSun, dzSun);
          if (distSun < minSolarDist) {
            const angle = distSun > 1e-4 ? Math.atan2(dzSun, dxSun) : (appState.data.stars.length + 1) * 1.15;
            x = Math.round((sunPos.x || 0) + Math.cos(angle) * minSolarDist);
            z = Math.round((sunPos.z || 0) + Math.sin(angle) * minSolarDist);
            adjusted = true;
          }
        }

        for (let pass = 0; pass < 4; pass++) {
          let collided = false;
          for (const other of appState.data.stars) {
            if (other.id === ignoreStarId) continue;
            const op = other.position || { x: 0, y: 0, z: 0 };
            const clearance = other.id === 'star-sun' ? minSolarDist : (Number(other.radius) || 14) + 120;
            const dx = x - (op.x || 0);
            const dz = z - (op.z || 0);
            const d = Math.hypot(dx, dz);
            if (d < clearance) {
              const ang = d > 1e-4 ? Math.atan2(dz, dx) : (pass + 1) * 1.57;
              x = Math.round((op.x || 0) + Math.cos(ang) * clearance);
              z = Math.round((op.z || 0) + Math.sin(ang) * clearance);
              collided = true;
              adjusted = true;
            }
          }
          if (!collided) break;
        }

        if (adjusted && notifyIfAdjusted) {
          showToast('太陽系内には恒星を配置できないため、外縁の安全領域へ移動しました', 'ph-shield-warning');
        }

        return { x: Math.round(x), y: Math.round(y), z: Math.round(z) };
      }

      function findSafeStarSpawnPosition() {
        const sun = getStarById('star-sun') || appState.data.stars[0];
        const sunPos = sun && sun.position ? sun.position : { x: 0, y: 0, z: 0 };
        const minDistanceFromSun = getSolarSystemOuterBoundary();
        const idx = appState.data.stars.length + 1;

        for (let attempt = 0; attempt < 48; attempt++) {
          const ring = Math.floor(attempt / 8);
          const angle = idx * 1.35 + attempt * 0.78;
          const dist = minDistanceFromSun + 45 + ring * 95;
          const candidate = {
            x: Math.round((sunPos.x || 0) + Math.cos(angle) * dist),
            y: Math.round((sunPos.y || 0) + (((attempt % 3) - 1) * 24)),
            z: Math.round((sunPos.z || 0) + Math.sin(angle) * dist)
          };

          const tooCloseToStar = appState.data.stars.some((s) => {
            const sp = s.position || { x: 0, y: 0, z: 0 };
            const d = Math.hypot(candidate.x - (sp.x || 0), candidate.z - (sp.z || 0));
            const requiredClearance = s.id === 'star-sun' ? minDistanceFromSun : (Number(s.radius) || 14) + 120;
            return d < requiredClearance;
          });

          const insideAnyOrbit = appState.data.planets.some((p) => {
            const pStar = getStarById(p.starId) || sun;
            const center = pStar && pStar.position ? pStar.position : { x: 0, y: 0, z: 0 };
            const d = Math.hypot(candidate.x - (center.x || 0), candidate.z - (center.z || 0));
            return d < Number(p.orbitRadius || 0) + 38;
          });

          if (!tooCloseToStar && !insideAnyOrbit) {
            return enforceStarExclusionRule(candidate, null, false);
          }
        }

        return enforceStarExclusionRule(
          {
            x: (sunPos.x || 0) + minDistanceFromSun + 90,
            y: 0,
            z: (sunPos.z || 0)
          },
          null,
          false
        );
      }

      function addNewStar() {
        const spawnPos = findSafeStarSpawnPosition();
        if (!spawnPos) {
          showToast('太陽系外縁に安全な配置スペースが見つかりません', 'ph-warning');
          return;
        }

        const idx = appState.data.stars.length + 1;
        const palette = ['#f59e0b', '#38bdf8', '#a855f7', '#10b981', '#f43f5e'];
        const color = palette[(idx - 1) % palette.length];
        const newStar = {
          id: `star-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
          name: `系外恒星 #${idx}`,
          color,
          radius: 13,
          position: spawnPos,
          orbitSpeedMultiplier: 1.0,
          note: '太陽系外縁の安全領域に生成された系外恒星です。'
        };
        appState.data.stars.push(newStar);
        appState.data.selectedKind = 'star';
        appState.data.selectedId = newStar.id;
        appState.data.lockedTarget = { kind: 'star', id: newStar.id };
        rebuild3DScene();
        renderAllUI();
        saveStateToLocalStorage(true);
        showToast(`太陽系外縁に恒星「${newStar.name}」を追加しました`, 'ph-sun');
      }

      function createPlanetFromForm(e) {
        e.preventDefault();
        const name = els.newPlanetName ? els.newPlanetName.value.trim() : '';
        if (!name) {
          if (els.newPlanetName) {
            els.newPlanetName.setCustomValidity('名称を入力して下さい');
            els.newPlanetName.reportValidity();
          }
          showToast('名称を入力して下さい', 'ph-warning');
          return;
        }
        if (els.newPlanetName) els.newPlanetName.setCustomValidity('');

        const starId = (els.newPlanetStar && els.newPlanetStar.value) || (appState.data.stars[0] && appState.data.stars[0].id);
        const type = els.newPlanetType && els.newPlanetType.value === 'image' ? 'image' : 'document';
        const radius = Number((els.newPlanetRadius && els.newPlanetRadius.value) || 3.8);
        const orbitRadius = Number((els.newPlanetOrbit && els.newPlanetOrbit.value) || 44);

        const defaultColors = {
          document: '#38bdf8',
          image: '#f472b6'
        };

        const planet = {
          id: `planet-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
          starId,
          name,
          type,
          color: defaultColors[type] || '#38bdf8',
          radius,
          orbitRadius,
          orbitSpeed: 0.24 + Math.random() * 0.2,
          orbitAngle: Math.random() * Math.PI * 2,
          inclination: (Math.random() - 0.5) * 0.16,
          contentHtml: `<h1>${escapeHtml(name)}</h1><p>新規作成されたドキュメント惑星です。</p>`,
          imageUrl:
            type === 'image'
              ? (els.newPlanetImageUrl && els.newPlanetImageUrl.value.trim()) ||
                createDefaultSampleImageDataUrl(name, '#312e81', '#db2777')
              : '',
          updatedAt: Date.now()
        };

        const file = els.newPlanetImageFile && els.newPlanetImageFile.files && els.newPlanetImageFile.files[0];
        if (type === 'image' && file) {
          const reader = new FileReader();
          reader.onload = () => {
            planet.imageUrl = String(reader.result || '');
            finalizeCreatedPlanet(planet);
          };
          reader.readAsDataURL(file);
        } else {
          finalizeCreatedPlanet(planet);
        }
      }

      function finalizeCreatedPlanet(planet) {
        appState.data.planets.push(planet);
        appState.data.selectedKind = 'planet';
        appState.data.selectedId = planet.id;
        appState.data.lockedTarget = { kind: 'planet', id: planet.id };
        if (!appState.data.expandedPlanetIds.includes(planet.id)) {
          appState.data.expandedPlanetIds.push(planet.id);
        }
        closeNewPlanetModal();
        rebuild3DScene();
        renderAllUI();
        saveStateToLocalStorage(true);
        showToast(`惑星「${planet.name}」を軌道に配置しました`, 'ph-planet');
      }

      function handleLocalFilesImport(filesList) {
        if (!filesList || filesList.length === 0) return;
        const targetStar = getStarById(appState.data.selectedId) || appState.data.stars[0];
        if (!targetStar) return;

        Array.from(filesList).forEach((file, index) => {
          const mime = (file.type || '').toLowerCase();
          const lowerName = (file.name || '').toLowerCase();
          let type = 'document';
          let color = '#38bdf8';

          if (mime.startsWith('image/') || /\.(png|jpg|jpeg|gif|webp|svg)$/.test(lowerName)) {
            type = 'image';
            color = '#f472b6';
          }

          const orbitRadius = 38 + ((appState.data.planets.length + index) % 6) * 20;
          const planet = {
            id: `planet-${Date.now()}-${index}-${Math.random().toString(36).slice(2, 5)}`,
            starId: targetStar.id,
            name: file.name,
            type,
            color,
            radius: 3.7,
            orbitRadius,
            orbitSpeed: 0.25 + index * 0.04,
            orbitAngle: index * 1.1,
            inclination: 0.04,
            contentHtml: `<h1>${escapeHtml(file.name)}</h1><p>ローカルファイルから読み込まれました（${Math.max(1, Math.round(file.size / 1024))} KB）。</p>`,
            imageUrl: '',
            updatedAt: Date.now()
          };

          if (type === 'image' && file.size <= 3 * 1024 * 1024) {
            const reader = new FileReader();
            reader.onload = () => {
              planet.imageUrl = String(reader.result || '');
              saveStateToLocalStorage(true);
            };
            reader.readAsDataURL(file);
          } else if (type === 'document' && file.size <= 512 * 1024) {
            const reader = new FileReader();
            reader.onload = () => {
              const txt = String(reader.result || '');
              planet.contentHtml = `<h1>${escapeHtml(file.name)}</h1><pre class="whitespace-pre-wrap text-xs bg-slate-950 p-3 rounded-xl border border-slate-800">${escapeHtml(txt.slice(0, 4000))}</pre>`;
              saveStateToLocalStorage(true);
            };
            reader.readAsText(file);
          }

          appState.data.planets.push(planet);
        });

        rebuild3DScene();
        renderAllUI();
        saveStateToLocalStorage(true);
        showToast(`${filesList.length} 件のファイルを惑星として配置しました`, 'ph-folder-open');
      }

      function openBlackHoleModal(kind, id) {
        if (kind === 'star') {
          if (appState.data.stars.length <= 1) {
            showToast('最後の恒星は消去できません', 'ph-warning');
            return;
          }
          const star = getStarById(id);
          if (!star) return;
          const childrenCount = getPlanetsForStar(star.id).length;
          appState.blackHoleTarget = { kind: 'star', id: star.id };
          if (els.blackHoleTargetName) els.blackHoleTargetName.textContent = `恒星: ${star.name}`;
          if (els.blackHoleDescription) {
            els.blackHoleDescription.textContent = `この恒星系と所属する ${childrenCount} 個の惑星をブラックホールへ消去します。この操作は元に戻せません。`;
          }
          setModalVisibility(els.blackHoleModal, true);
        } else if (kind === 'planet') {
          const planet = getPlanetById(id);
          if (!planet) return;
          appState.blackHoleTarget = { kind: 'planet', id: planet.id };
          if (els.blackHoleTargetName) els.blackHoleTargetName.textContent = `惑星: ${planet.name}`;
          if (els.blackHoleDescription) {
            els.blackHoleDescription.textContent = `惑星ファイル「${planet.name}」をブラックホールへ消去します。よろしいですか？`;
          }
          setModalVisibility(els.blackHoleModal, true);
        }
      }

      function closeBlackHoleModal() {
        appState.blackHoleTarget = null;
        setModalVisibility(els.blackHoleModal, false);
      }

      function confirmBlackHoleDeletion() {
        const target = appState.blackHoleTarget;
        if (!target) return;

        if (target.kind === 'star') {
          appState.data.stars = appState.data.stars.filter((s) => s.id !== target.id);
          appState.data.planets = appState.data.planets.filter((p) => p.starId !== target.id);
          appState.data.selectedKind = 'star';
          appState.data.selectedId = appState.data.stars[0] ? appState.data.stars[0].id : null;
          if (appState.data.lockedTarget && appState.data.lockedTarget.id === target.id) {
            appState.data.lockedTarget = null;
          }
          showToast('恒星系をブラックホールへ消去しました', 'ph-trash');
        } else if (target.kind === 'planet') {
          appState.data.planets = appState.data.planets.filter((p) => p.id !== target.id);
          if (appState.data.selectedKind === 'planet' && appState.data.selectedId === target.id) {
            appState.data.selectedKind = 'star';
            appState.data.selectedId = appState.data.stars[0] ? appState.data.stars[0].id : null;
          }
          if (appState.data.lockedTarget && appState.data.lockedTarget.id === target.id) {
            appState.data.lockedTarget = null;
          }
          showToast('惑星をブラックホールへ消去しました', 'ph-trash');
        }

        closeBlackHoleModal();
        reconcileSelection();
        rebuild3DScene();
        renderAllUI();
        saveStateToLocalStorage(true);
      }

      function openDocEditor(planetId) {
        const planet = getPlanetById(planetId);
        if (!planet) return;

        appState.activeDocPlanetId = planet.id;

        if (els.docSaveIndicator) els.docSaveIndicator.textContent = '自動保存済み';
        setModalVisibility(els.astralDocModal, true);
        if (els.astralDocEditor) els.astralDocEditor.focus();
      }

      function saveDocEditorContent(closeAfter = false) {
        if (!appState.activeDocPlanetId) return;
        const planet = getPlanetById(appState.activeDocPlanetId);
        if (planet && els.astralDocEditor) {
          planet.contentHtml = els.astralDocEditor.innerHTML;
          planet.updatedAt = Date.now();
          saveStateToLocalStorage(true);
          if (els.docSaveIndicator) els.docSaveIndicator.textContent = '自動保存済み';
        }
        if (closeAfter) {
          setModalVisibility(els.astralDocModal, false);
          appState.activeDocPlanetId = null;
          renderExplorer();
        }
      }

      function openImageViewer(planetId) {
        const planet = getPlanetById(planetId);
        if (!planet) return;
        const star = getStarById(planet.starId);
        appState.activeImagePlanetId = planet.id;
        appState.imageViewer.scale = 1;
        appState.imageViewer.rotation = 0;

        if (!planet.imageUrl) {
          planet.imageUrl = createDefaultSampleImageDataUrl(planet.name, '#1e3a8a', '#7e22ce');
          saveStateToLocalStorage(false);
        }

        if (els.imageViewerTitle) els.imageViewerTitle.textContent = planet.name;
        if (els.imageViewerSubtitle) els.imageViewerSubtitle.textContent = `所属恒星: ${star ? star.name : '未所属'}`;
        if (els.imageViewerUrlInput) els.imageViewerUrlInput.value = planet.imageUrl || '';
        if (els.imageViewerError) els.imageViewerError.classList.add('hidden');
        if (els.imageViewerImg) els.imageViewerImg.src = planet.imageUrl || '';
        updateImageViewerTransform();
        setModalVisibility(els.astralImageViewerModal, true);
      }

      function updateImageViewerTransform() {
        const { scale, rotation } = appState.imageViewer;
        if (els.imageViewerImg) {
          els.imageViewerImg.style.transform = `scale(${scale}) rotate(${rotation}deg)`;
        }
        if (els.imageViewerZoomLabel) {
          els.imageViewerZoomLabel.textContent = `${Math.round(scale * 100)}%`;
        }
      }

      // =========================================================================
      // 11. 検索ドロップダウン & HUD表示切替
      // =========================================================================
      function renderSearchDropdown() {
        const q = appState.searchQuery.trim().toLowerCase();
        if (els.btnClearSearch) els.btnClearSearch.classList.toggle('hidden', q.length === 0);
        if (!els.searchDropdown) return;
        if (!q) {
          els.searchDropdown.classList.add('hidden');
          return;
        }

        const matchedStars = appState.data.stars.filter((s) => String(s.name || '').toLowerCase().includes(q));
        const matchedPlanets = appState.data.planets.filter((p) => String(p.name || '').toLowerCase().includes(q));

        els.searchDropdown.innerHTML = '';
        if (matchedStars.length === 0 && matchedPlanets.length === 0) {
          els.searchDropdown.innerHTML = '<div class="px-3 py-2 text-slate-500">該当する恒星・惑星がありません</div>';
          els.searchDropdown.classList.remove('hidden');
          return;
        }

        matchedStars.forEach((s) => {
          const row = document.createElement('button');
          row.type = 'button';
          row.className = 'w-full flex items-center gap-2 rounded-lg px-2.5 py-1.5 text-left hover:bg-slate-800 text-amber-200';
          row.innerHTML = `<i class="ph-fill ph-sun text-amber-400"></i><span class="truncate">${escapeHtml(s.name)}</span>`;
          row.addEventListener('click', () => {
            selectEntity('star', s.id, true, true);
            els.searchDropdown.classList.add('hidden');
          });
          els.searchDropdown.appendChild(row);
        });

        matchedPlanets.forEach((p) => {
          const badge = getTypeBadge(normalizePlanetType(p));
          const row = document.createElement('button');
          row.type = 'button';
          row.className = 'w-full flex items-center gap-2 rounded-lg px-2.5 py-1.5 text-left hover:bg-slate-800 text-slate-200';
          row.innerHTML = `<i class="ph ${badge.icon} ${badge.color}"></i><span class="truncate">${escapeHtml(p.name)}</span>`;
          row.addEventListener('click', () => {
            selectEntity('planet', p.id, true, false);
            els.searchDropdown.classList.add('hidden');
          });
          els.searchDropdown.appendChild(row);
        });

        els.searchDropdown.classList.remove('hidden');
      }

            return;
          }
          appState.syncConfig = {
            enabled: true,
            url: fields.url,
            anonKey: fields.anonKey,
            table: fields.table,
            roomId: fields.roomId
          };
          saveSyncConfigToLocalStorage();
          updateSyncStatusUI('接続中...', 'bg-amber-400');
          closeSyncModal();
          startSyncLoop();
          await pullStateFromSupabase(true);
        });

        onEl(els.btnDisconnectSync, 'click', () => {
          stopSyncLoop();
          appState.isSyncing = false;
          appState.syncConfig.enabled = false;
          saveSyncConfigToLocalStorage();
          updateSyncStatusUI('ローカル保存（未接続）', 'bg-slate-500');
          showToast('Supabase同期を切断しました', 'ph-cloud-slash');
        });

        onEl(els.btnSyncNow, 'click', async () => {
          if (!isInviteAuthenticated()) {
            const codeVal = els.inviteCodeInput ? els.inviteCodeInput.value : '';
            const ok = await verifyAndAuthenticateInviteCode(codeVal, true);
            if (!ok) return;
          }
          const fields = readSyncFormFields();
          if (!fields.url || !fields.anonKey) {
            showToast('先に URL と Anon Key を入力してください', 'ph-warning');
            return;
          }
          appState.syncConfig = {
            enabled: true,
            url: fields.url,
            anonKey: fields.anonKey,
            table: fields.table,
            roomId: fields.roomId
          };
          saveSyncConfigToLocalStorage();
          if (!appState.syncTimerId) {
            startSyncLoop();
          } else {
            await pullStateFromSupabase(true);
          }
        });

        onEl(els.btnCancelBlackHole, 'click', closeBlackHoleModal);
        onEl(els.btnConfirmBlackHole, 'click', confirmBlackHoleDeletion);

        els.docCmdBtns.forEach((btn) => {
          onEl(btn, 'click', () => {
            if (els.astralDocEditor) els.astralDocEditor.focus();
            const cmd = btn.dataset.docCmd;
            const block = btn.dataset.docBlock;
            try {
              if (cmd) {
                document.execCommand(cmd, false, null);
              } else if (block) {
                document.execCommand('formatBlock', false, block);
              }
            } catch (err) {
              // 非対応環境フォールバック
            }
            if (els.docSaveIndicator) els.docSaveIndicator.textContent = '未保存の変更あり';
          });
        });

        onEl(els.astralDocEditor, 'input', () => {
          if (els.docSaveIndicator) els.docSaveIndicator.textContent = '保存中...';
          if (appState.docAutoSaveTimerId) clearTimeout(appState.docAutoSaveTimerId);
          appState.docAutoSaveTimerId = setTimeout(() => {
            saveDocEditorContent(false);
          }, 500);
        });

        onEl(els.btnSaveAndCloseDoc, 'click', () => saveDocEditorContent(true));

        onEl(els.btnViewerZoomIn, 'click', () => {
          appState.imageViewer.scale = Math.min(5, Number((appState.imageViewer.scale + 0.25).toFixed(2)));
          updateImageViewerTransform();
        });
        onEl(els.btnViewerZoomOut, 'click', () => {
          appState.imageViewer.scale = Math.max(0.25, Number((appState.imageViewer.scale - 0.25).toFixed(2)));
          updateImageViewerTransform();
        });
        onEl(els.btnViewerFit, 'click', () => {
          appState.imageViewer.scale = 1;
          updateImageViewerTransform();
        });
        onEl(els.btnViewerReset, 'click', () => {
          appState.imageViewer.scale = 1;
          appState.imageViewer.rotation = 0;
          updateImageViewerTransform();
        });
        onEl(els.btnViewerRotate, 'click', () => {
          appState.imageViewer.rotation = (appState.imageViewer.rotation + 90) % 360;
          updateImageViewerTransform();
        });
        onEl(els.btnCloseImageViewer, 'click', () => {
          setModalVisibility(els.astralImageViewerModal, false);
          appState.activeImagePlanetId = null;
        });
        onEl(els.imageViewerImg, 'error', () => {
          if (els.imageViewerError) els.imageViewerError.classList.remove('hidden');
        });
        onEl(els.imageViewerImg, 'load', () => {
          if (els.imageViewerError) els.imageViewerError.classList.add('hidden');
        });
        onEl(els.btnApplyViewerUrl, 'click', () => {
          const planet = getPlanetById(appState.activeImagePlanetId);
          if (!planet || !els.imageViewerUrlInput) return;
          const url = els.imageViewerUrlInput.value.trim();
          if (!url) return;
          planet.imageUrl = url;
          planet.updatedAt = Date.now();
          if (els.imageViewerImg) els.imageViewerImg.src = url;
          saveStateToLocalStorage(true);
          showToast('画像URLを保存しました', 'ph-image');
        });
        onEl(els.btnUploadViewerFile, 'click', () => {
          if (els.imageViewerFileInput) els.imageViewerFileInput.click();
        });
        onEl(els.imageViewerFileInput, 'change', (e) => {
          const file = e.target.files && e.target.files[0];
          const planet = getPlanetById(appState.activeImagePlanetId);
          if (!file || !planet) return;
          const reader = new FileReader();
          reader.onload = () => {
            const dataUrl = String(reader.result || '');
            planet.imageUrl = dataUrl;
            planet.updatedAt = Date.now();
            if (els.imageViewerUrlInput) els.imageViewerUrlInput.value = dataUrl;
            if (els.imageViewerImg) els.imageViewerImg.src = dataUrl;
            saveStateToLocalStorage(true);
            showToast('端末画像を惑星に保存しました', 'ph-image');
          };
          reader.readAsDataURL(file);
          e.target.value = '';
        });
      }

      let booted = false;
      function bootAstralOcean() {
        if (booted) return;
        booted = true;
        try {
          reconcileSelection();
          initViewportEngine();
          bindAllEvents();
          renderAllUI();
          if (appState.syncConfig && appState.syncConfig.enabled && isInviteAuthenticated()) {
            startSyncLoop();
          }
        } catch (err) {
          try {
            if (!viewport.canvas2D && els.viewportContainer) {
              const c = document.createElement('canvas');
              c.width = els.viewportContainer.clientWidth || window.innerWidth || 1024;
              c.height = els.viewportContainer.clientHeight || window.innerHeight || 768;
              viewport.canvas2D = c;
              viewport.ctx2D = c.getContext('2d');
              els.viewportContainer.appendChild(c);
            }
            appState.data.viewMode = '2d';
            applyViewModeVisibility();
            startRenderLoop();
          } catch (innerErr) {
            // 無視
          }
        }
      }

      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', bootAstralOcean);
      } else {
        bootAstralOcean();
      }
      window.addEventListener('load', () => {
        bootAstralOcean();
        if (appState.data.viewMode === '3d' && !viewport.renderer && ensureThreeSetup()) {
          applyViewModeVisibility();
          rebuild3DScene();
        }
      });
    })();
  </script>
